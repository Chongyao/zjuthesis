\begin{algorithm}[htbp]
  \caption{自适应位移：初始化阶段}\label{alg:adaptive-shifting}
  \begin{algorithmic}[1]
    \STATE $\mu \gets 0,\ \sigma \gets 0$
    \STATE 求解 \cref{eq:final-eig}，记录 $\lambda_{\min}$ 和 $\lambda_{\text{cut-off}}$
    \IF{失败}
      \STATE $\mu \gets 10^{-3},\ \sigma \gets 10^{-6}$
    \ELSE
      \RETURN 特征值和特征向量
    \ENDIF
    \STATE 求解 \cref{eq:final-eig}，记录 $\lambda_{\min}$ 和 $\lambda_{\text{cut-off}}$
    \WHILE{失败}
      \STATE $\mu \gets 10\mu,\ \sigma \gets 10\sigma$
      \STATE 求解 \cref{eq:final-eig}，记录 $\lambda_{\min}$ 和 $\lambda_{\text{cut-off}}$
    \ENDWHILE
  \end{algorithmic}
\end{algorithm}

\begin{algorithm}[htbp]
  \caption{自适应位移：自适应回退阶段}\label{alg:adaptive-shifting-refine}
  \begin{algorithmic}[1]
    \STATE $\alpha \gets 4,\ \beta \gets 10^{-6}$
    \WHILE{$\mu \ge \beta \lambda_{\min}$}
      \STATE $\mu \gets \mu / 10$
      \STATE 求解 \cref{eq:final-eig}，记录 $\lambda_{\min}$ 和 $\lambda_{\text{cut-off}}$
      \IF{异常}
        \STATE $\mu \gets 10\mu$，\textbf{break}
      \ENDIF
    \ENDWHILE
    \WHILE{$\mu / \sigma \le \alpha \lambda_{\text{cut-off}}$}
      \STATE $\sigma \gets \sigma / 10$
      \STATE 求解 \cref{eq:final-eig}，记录 $\lambda_{\min}$ 和 $\lambda_{\text{cut-off}}$
      \IF{异常}
        \STATE $\sigma \gets 10\sigma$，\textbf{break}
      \ENDIF
    \ENDWHILE
    \RETURN 特征值和特征向量
  \end{algorithmic}
\end{algorithm}
