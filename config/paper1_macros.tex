% TikZ-based matrix pattern operators (require tikz package)
\DeclareMathOperator{\LowerTri}{
\begin{tikzpicture}[baseline]
\draw (0,-0.2em) -- (0em,0.8em) -- (1em,-0.2em) -- (0,-0.2em);
\node at (0.25em,0.1em) {\tiny{$\I$}};
\end{tikzpicture}
}

\begin{tikzpicture}[baseline]
\draw (0,0.8em) -- (1em,0.8em) -- (1em,-0.2em) -- (0,0.8em);

\end{tikzpicture}
}

  \draw (0,0.6em) -- (0.2em, 0.8em) -- (1em, 0) -- (0.8em, -0.2em) -- (0, 0.6em);
\end{tikzpicture}
}
\DeclareMathOperator{\TriDiag} {

  \draw (0.1em, 0.6em) -- (0.8em, -0.1em);
